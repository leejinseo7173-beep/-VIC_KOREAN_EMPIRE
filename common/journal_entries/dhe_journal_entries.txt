je_dhe_susinsa_bobingsa_mission = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = dhe_reforms

	is_shown_in_lobby = {
		country_has_primary_culture = cu:korean
	}

	is_shown_when_inactive = {
		exists = c:KOR
		this = c:KOR
		has_variable = dhe_gwangmu_reform_started
		has_variable = dhe_susinsa_bobingsa_unlocked
		NOT = { has_variable = dhe_susinsa_bobingsa_done }
		NOT = { has_variable = dhe_gwangmu_reform_failed }
	}

	possible = {
		exists = c:KOR
		this = c:KOR
		has_variable = dhe_gwangmu_reform_started
		has_variable = dhe_susinsa_bobingsa_unlocked
		NOT = { has_variable = dhe_susinsa_bobingsa_done }
		NOT = { has_variable = dhe_gwangmu_reform_failed }
	}

	invalid = {
		NOT = { has_variable = dhe_susinsa_bobingsa_unlocked }
	}

	scripted_progress_bar = dhe_progress_bar_susinsa_bobingsa

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = dhe_mission_status_returned_home
				trigger = {
					has_variable = dhe_mission_returned_home
					exists = var:dhe_mission_leader_storage_var
				}
			}
			triggered_desc = {
				desc = dhe_mission_status_returning_home
				trigger = {
					has_variable = dhe_mission_en_route_home
					exists = var:dhe_mission_leader_storage_var
				}
			}
			triggered_desc = {
				desc = dhe_mission_status_abroad
				trigger = {
					exists = var:dhe_mission_current_location_var
					exists = var:dhe_mission_leader_storage_var
				}
			}
			triggered_desc = {
				desc = dhe_mission_status_abroad_no_leader
				trigger = {
					exists = var:dhe_mission_current_location_var
					NOT = { exists = var:dhe_mission_leader_storage_var }
				}
			}
			triggered_desc = {
				desc = dhe_mission_status_leader_selected
				trigger = {
					exists = var:dhe_mission_leader_storage_var
				}
			}
			triggered_desc = {
				desc = dhe_mission_status_returned_home_no_leader
				trigger = {
					has_variable = dhe_mission_returned_home
				}
			}
			triggered_desc = {
				desc = dhe_mission_status_waiting
				trigger = { always = yes }
			}
		}
	}

	immediate = {
		add_modifier = { name = dhe_mod_susinsa_bobingsa_active months = -1 }
		set_variable = { name = dhe_mission_learning_score value = 0 }
		trigger_event = { id = dhe_mission_event.1 popup = yes }
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = dhe_mission_current_location_var
				}
				change_variable = { name = dhe_mission_learning_score add = 2 }
				je:je_dhe_susinsa_bobingsa_mission ?= {
					add_progress = { name = dhe_progress_bar_susinsa_bobingsa value = 3 }
				}
			}
			if = {
				limit = {
					exists = var:dhe_mission_current_location_var
					var:dhe_mission_current_location_var.techs_researched > ROOT.techs_researched
				}
				change_variable = { name = dhe_mission_learning_score add = 2 }
				je:je_dhe_susinsa_bobingsa_mission ?= {
					add_progress = { name = dhe_progress_bar_susinsa_bobingsa value = 2 }
				}
			}
			if = {
				limit = {
					exists = c:GBR
					c:GBR = {
						country_rank >= rank_value:major_power
						relations:root >= relations_threshold:cordial
					}
				}
				set_variable = { name = dhe_mission_british_exchange days = 45 }
			}
			if = {
				limit = {
					exists = c:FRA
					c:FRA = {
						country_rank >= rank_value:major_power
						relations:root >= relations_threshold:cordial
					}
				}
				set_variable = { name = dhe_mission_french_exchange days = 45 }
			}
			if = {
				limit = {
					exists = c:USA
					c:USA = {
						country_rank >= rank_value:major_power
						relations:root >= relations_threshold:cordial
					}
				}
				set_variable = { name = dhe_mission_american_exchange days = 45 }
			}
			if = {
				limit = {
					exists = c:RUS
					c:RUS = {
						country_rank >= rank_value:major_power
						relations:root >= relations_threshold:cordial
					}
				}
				set_variable = { name = dhe_mission_russian_exchange days = 45 }
			}
			if = {
				limit = {
					OR = {
						AND = {
							exists = c:GER
							c:GER = {
								country_rank >= rank_value:major_power
								relations:root >= relations_threshold:cordial
							}
						}
						AND = {
							exists = c:PRU
							c:PRU = {
								country_rank >= rank_value:major_power
								relations:root >= relations_threshold:cordial
							}
						}
					}
				}
				set_variable = { name = dhe_mission_german_exchange days = 45 }
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = dhe_mission_returned_home_tt
			has_variable = dhe_mission_returned_home
		}
		custom_tooltip = {
			text = dhe_mission_complete_report_tt
			has_variable = dhe_mission_complete_report
		}
		custom_tooltip = {
			text = dhe_mission_japan_report_tt
			has_variable = dhe_mission_japan_report
		}
		custom_tooltip = {
			text = dhe_mission_qing_report_tt
			has_variable = dhe_mission_qing_report
		}
		custom_tooltip = {
			text = dhe_mission_west_report_tt
			has_variable = dhe_mission_west_report
		}
		custom_tooltip = {
			text = dhe_mission_progress_complete_tt
			je:je_dhe_susinsa_bobingsa_mission ?= {
				"scripted_bar_progress(dhe_progress_bar_susinsa_bobingsa)" >= 100
			}
		}
	}

	on_complete = {
		set_variable = dhe_susinsa_bobingsa_done
		remove_modifier = dhe_mod_susinsa_bobingsa_active
		var:dhe_mission_leader_storage_var ?= {
			set_character_busy_and_immortal = no
		}
		add_modifier = { name = dhe_mod_mission_reform_reports months = 120 }
		if = {
			limit = { has_variable = dhe_mission_british_exchange }
			add_modifier = { name = dhe_mod_british_technical_exchange months = 60 }
		}
		if = {
			limit = { has_variable = dhe_mission_french_exchange }
			add_modifier = { name = dhe_mod_french_legal_exchange months = 60 }
		}
		if = {
			limit = { has_variable = dhe_mission_american_exchange }
			add_modifier = { name = dhe_mod_american_education_exchange months = 60 }
		}
		if = {
			limit = { has_variable = dhe_mission_russian_exchange }
			add_modifier = { name = dhe_mod_russian_border_exchange months = 60 }
		}
		if = {
			limit = { has_variable = dhe_mission_german_exchange }
			add_modifier = { name = dhe_mod_german_military_exchange months = 60 }
		}
		trigger_event = { id = dhe_mission_event.5 popup = yes }
		je:je_dhe_gwangmu_reform_main ?= {
			add_progress = { name = dhe_progress_bar_gwangmu value = 10 }
		}
	}
}

je_korea_capital_industry = {
	icon = "gfx/interface/icons/event_icons/dhe_hanjiang_qiji.dds"
	group = dhe_reforms
	is_shown_in_lobby = { country_has_primary_culture = cu:korean }
	is_shown_when_inactive = {
		exists = c:KOR
		this = c:KOR
		has_variable = dhe_gwangmu_reform_started
		NOT = { has_variable = korea_capital_industry_done }
	}

	possible = {
		exists = c:KOR
		this = c:KOR
		has_variable = dhe_gwangmu_reform_started
		NOT = { has_variable = korea_capital_industry_done }
		
		any_scope_state = {
			owner = ROOT
			OR = { state_region = s:STATE_SEOUL state_region = s:STATE_HAESEO }
		}
	}

	complete = {
		custom_tooltip = { text = korea_capital_industry_goal_tt }
		any_scope_state = {
			state_region = s:STATE_SEOUL
			owner = ROOT
			any_scope_building = { is_building_type = building_urban_center level >= 10 }
			any_scope_building = { is_building_type = building_government_administration level >= 5 }
			any_scope_building = { is_building_type = building_university level >= 5 }
			any_scope_building = { is_building_type = building_railway level >= 5 } 
			any_scope_building = { is_building_type = building_tooling_workshop level >= 5 }
			any_scope_building = { is_building_type = building_steel_mill level >= 5 }
			any_scope_building = { is_building_type = building_motor_industry level >= 3 } # 전동기 공장
			any_scope_building = { is_building_type = building_arms_industry level >= 3 }  # 무기 공장
		}
		any_scope_state = {
			state_region = s:STATE_HAESEO
			owner = ROOT
			any_scope_building = { is_building_type = building_university level >= 2 }
			any_scope_building = { is_building_type = building_paper_mill level >= 5 }
			any_scope_building = { is_building_type = building_glassworks level >= 5 }
			any_scope_building = { is_building_type = building_iron_mine level >= 5 } 
			any_scope_building = { is_building_type = building_chemical_plant level >= 3 } # 화학 공장
		}
	}
	
	on_complete = {
		set_variable = korea_capital_industry_done
		every_scope_state = {
			limit = {
				owner = ROOT
				OR = { state_region = s:STATE_SEOUL state_region = s:STATE_HAESEO }
			}
			if = { limit = { has_modifier = modifier_capital_developing } remove_modifier = modifier_capital_developing }
			add_modifier = { name = modifier_capital_completed months = 120 }
		}
		# 💥 [복구 완료] 메인 일지에 진척도 10 쏴주기!
		je:je_dhe_gwangmu_reform_main ?= {
			add_progress = { name = dhe_progress_bar_gwangmu value = 10 }
		}
	}
}
