		first_valid = {
			triggered_desc = {

	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = dhe_conquest

	is_shown_in_lobby = { c:KOR ?= THIS }
	is_shown_when_inactive = {
		c:KOR ?= THIS
		has_variable = dhe_independence_achieved
		NOT = { has_variable = dhe_imperial_foreign_strategy_resolved }
	}

	possible = {
		c:KOR ?= THIS
		has_variable = dhe_independence_achieved
		NOT = { has_variable = dhe_imperial_foreign_strategy_resolved }
	}

	immediate = {
		scope:journal_entry ?= {
			add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 0 }
		}
		if = {
			limit = { has_variable = dhe_manchuria_influence_resolved }
			je:je_dhe_imperial_foreign_strategy ?= {
				add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 20 }
			}
		}
		if = {
			limit = { has_variable = dhe_jiandao_resolved }
			je:je_dhe_imperial_foreign_strategy ?= {
				add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 25 }
			}
		}
		if = {
			limit = { has_variable = dhe_jiandao_arable_resolved }
			je:je_dhe_imperial_foreign_strategy ?= {
				add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 15 }
			}
		}
		if = {
			limit = { has_variable = dhe_outer_manchuria_resolved }
			je:je_dhe_imperial_foreign_strategy ?= {
				add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 40 }
			}
		}
		if = {
			limit = { has_variable = dhe_korea_japan_trade_treaty_signed }
			je:je_dhe_imperial_foreign_strategy ?= {
				add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 30 }
			}
		}
		if = {
			limit = { has_variable = dhe_japan_control_resolved }
			je:je_dhe_imperial_foreign_strategy ?= {
				add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 40 }
			}
		}
		if = {
			limit = { NOT = { has_variable = dhe_foreign_strategy_meeting_called } }
			set_variable = dhe_foreign_strategy_meeting_called
			trigger_event = { id = dhe_expansion.5 days = 1 popup = yes }
		}
	}

	scripted_progress_bar = dhe_progress_bar_imperial_foreign_strategy

	widget = {
		gui = "gui/journal_entry_widgets/dhe_foreign_strategy_widgets.gui"
		name = "widget_je_dhe_imperial_foreign_strategy_header"
		container = "custom_widget_container_3"
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = dhe_imperial_foreign_strategy_status_northern_done
				trigger = {
					has_variable = dhe_outer_manchuria_resolved
					NOT = { has_journal_entry = je_dhe_southern_campaign_doctrine }
					NOT = { has_variable = dhe_southern_campaign_doctrine_resolved }
				}
			}
			triggered_desc = {
				desc = dhe_imperial_foreign_strategy_status_northern_active
				trigger = {
					OR = {
						has_journal_entry = je_dhe_northern_campaign_doctrine
						has_journal_entry = je_dhe_manchuria_rivalry
						has_journal_entry = je_dhe_jiandao_expansion
						has_journal_entry = je_dhe_outer_manchuria
					}
				}
			}
			triggered_desc = {
				desc = dhe_imperial_foreign_strategy_status_southern_active
				trigger = {
					OR = {
						has_journal_entry = je_dhe_southern_campaign_doctrine
						has_journal_entry = je_dhe_gwangmuho_incident
						has_journal_entry = je_dhe_japan_control
						has_variable = dhe_korea_japan_trade_treaty_signed
					}
				}
			}
			triggered_desc = {
				desc = dhe_imperial_foreign_strategy_status_waiting
				trigger = { always = yes }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = dhe_northern_campaign_doctrine_resolved
					NOT = { has_variable = dhe_second_strategy_meeting_called }
					NOT = { has_variable = dhe_chose_southern_strategy }
				}
				set_variable = dhe_second_strategy_meeting_called
				trigger_event = { id = dhe_expansion.10 days = 1 popup = yes }
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = dhe_imperial_foreign_strategy_progress_max_tt
			tooltip = dhe_imperial_foreign_strategy_progress_max_desc_tt
			je:je_dhe_imperial_foreign_strategy ?= {
				"scripted_bar_progress(dhe_progress_bar_imperial_foreign_strategy)" >= 100
			}
		}
		custom_tooltip = {
			text = dhe_imperial_foreign_strategy_second_route_tt
			tooltip = dhe_imperial_foreign_strategy_second_route_desc_tt
			OR = {
				has_variable = dhe_southern_campaign_doctrine_resolved
				has_variable = dhe_japan_control_resolved
			}
		}
	}

	on_complete = {
		set_variable = dhe_imperial_foreign_strategy_resolved
		add_prestige = 50
	}
}

		}
		set_variable = {
			name = dhe_jiandao_crop_counter_var
			value = 0
		}
	}

	scripted_progress_bar = dhe_progress_bar_jiandao_arable
	scripted_button = dhe_button_jiandao_arable_land
	scripted_button = dhe_button_jiandao_crop_potential

	complete = {
		custom_tooltip = {
			text = dhe_jiandao_settlement_goal_tt
			s:STATE_JIANDAO.region_state:KOR = {
				state_population > 100000
				gdp > 100000
			}
		}
		custom_tooltip = {
			text = dhe_jiandao_arable_land_goal_tt
			je:je_dhe_jiandao_arable_expansion ?= {
				"scripted_bar_progress(dhe_progress_bar_jiandao_arable)" >= 3
			}
		}
		custom_tooltip = {
			text = dhe_jiandao_crop_potential_goal_tt
			exists = var:dhe_jiandao_crop_counter_var
			var:dhe_jiandao_crop_counter_var >= 4
		}
	}

	on_complete = {
		set_variable = dhe_jiandao_arable_resolved
		je:je_dhe_imperial_foreign_strategy ?= {
			add_progress = { name = dhe_progress_bar_imperial_foreign_strategy value = 15 }
		}
		s:STATE_JIANDAO.region_state:KOR = {
			add_modifier = {
				name = dhe_mod_black_soil_development
				months = 120
			}
		}
		remove_variable = dhe_jiandao_arable_counter_var
		remove_variable = dhe_jiandao_crop_counter_var
		remove_variable = dhe_jiandao_rice_trial_var
		remove_variable = dhe_jiandao_dye_trial_var
		remove_variable = dhe_jiandao_tobacco_trial_var
		remove_variable = dhe_jiandao_ginseng_trial_var
		trigger_event = { id = dhe_jiandao.7 popup = yes }
	}
}
