	desc = "dhe_button_ic_public_speech_desc"

		has_journal_entry = je_dhe_independence_club
	}

	possible = {
		authority >= 50
		ig:ig_intelligentsia = { clout >= 0.10 }

		set_variable = { name = dhe_cool_ic_public_speech days = 180 }
		set_variable = { name = dhe_ic_public_network days = 365 }
		ig:ig_intelligentsia = { add_ig_approval = 1 }
	}
}

dhe_button_jiandao_arable_land = {
	name = "dhe_button_jiandao_arable_land"
	desc = "dhe_button_jiandao_arable_land_desc"

	cooldown = {
		days = 1095
	}

	visible = {
		has_journal_entry = je_dhe_jiandao_arable_expansion
		var:dhe_jiandao_arable_counter_var < 3
	}

	possible = {
		gold_reserves >= 25000
		owns_entire_state_region = STATE_JIANDAO
	}

	effect = {
		change_variable = {
			name = dhe_jiandao_arable_counter_var
			add = 1
		}
		je:je_dhe_jiandao_arable_expansion ?= {
			add_progress = { name = dhe_progress_bar_jiandao_arable value = 1 }
		}
		trigger_event = { id = dhe_jiandao.5 popup = yes }
		show_as_tooltip = {
			s:STATE_JIANDAO.region_state:KOR = {
				state_region = {
					add_arable_land = 10
				}
			}
		}
	}

	ai_chance = {
		value = 10
	}
}
