dhe_jungchuwon_apply_political_basis_drift = {

			any_interest_group = {
				ig_clout >= 0.25

						has_ideology = ideology:ideology_reformer

						has_ideology = ideology:ideology_liberal_leader

						has_ideology = ideology:ideology_humanitarian
					}
				}
			}
		}

			add_progress = { value = 9 name = dhe_progress_bar_jungchuwon_reformists }
			add_progress = { value = -5 name = dhe_progress_bar_jungchuwon_royalists }
			add_progress = { value = -4 name = dhe_progress_bar_jungchuwon_provincials }
		}
	}

		limit = {
			any_interest_group = {
				ig_clout >= 0.15

			add_progress = { value = 6 name = dhe_progress_bar_jungchuwon_reformists }

			add_progress = { value = -3 name = dhe_progress_bar_jungchuwon_provincials }
		}
	}

	if = {
		limit = {
			any_interest_group = {
				ig_clout >= 0.25
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_royalist

						has_ideology = ideology:ideology_paternalistic
						has_ideology = ideology:ideology_reactionary
						has_ideology = ideology:ideology_developmental_autocrat
					}
				}
			}
		}
		je:je_dhe_jungchuwon_system ?= {
			add_progress = { value = 9 name = dhe_progress_bar_jungchuwon_royalists }
			add_progress = { value = -5 name = dhe_progress_bar_jungchuwon_reformists }
			add_progress = { value = -4 name = dhe_progress_bar_jungchuwon_provincials }
		}

			any_interest_group = {
				ig_clout >= 0.15
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_royalist

						has_ideology = ideology:ideology_paternalistic
						has_ideology = ideology:ideology_reactionary
						has_ideology = ideology:ideology_developmental_autocrat
					}
				}
			}
		}
		je:je_dhe_jungchuwon_system ?= {
			add_progress = { value = 6 name = dhe_progress_bar_jungchuwon_royalists }
			add_progress = { value = -3 name = dhe_progress_bar_jungchuwon_reformists }
			add_progress = { value = -3 name = dhe_progress_bar_jungchuwon_provincials }
		}

		limit = {
			any_interest_group = {
				ig_clout >= 0.20
				OR = {
					is_interest_group_type = ig_rural_folk
					is_interest_group_type = ig_petty_bourgeoisie
					is_interest_group_type = ig_industrialists

					OR = {
						has_ideology = ideology:ideology_moderate
						has_ideology = ideology:ideology_protectionist
						has_ideology = ideology:ideology_particularist
						has_ideology = ideology:ideology_agrarian
					}
				}
			}
		}
		je:je_dhe_jungchuwon_system ?= {
			add_progress = { value = 6 name = dhe_progress_bar_jungchuwon_provincials }
			add_progress = { value = -3 name = dhe_progress_bar_jungchuwon_royalists }
			add_progress = { value = -3 name = dhe_progress_bar_jungchuwon_reformists }
