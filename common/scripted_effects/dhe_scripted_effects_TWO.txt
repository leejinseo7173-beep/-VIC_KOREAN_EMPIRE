dhe_samjeong_init_unrest_gauge = {
	if = {
		limit = { NOT = { has_variable = dhe_samjeong_unrest_gauge } }
		set_variable = {
			name = dhe_samjeong_unrest_gauge
			value = 350
		}
	}
	dhe_samjeong_resolve_unrest_gauge = yes
}

dhe_samjeong_resolve_unrest_gauge = {
	if = {
		limit = { var:dhe_samjeong_unrest_gauge > 1000 }
		set_variable = {
			name = dhe_samjeong_unrest_gauge
			value = 1000
		}
	}
	if = {
		limit = { var:dhe_samjeong_unrest_gauge < 0 }
		set_variable = {
			name = dhe_samjeong_unrest_gauge
			value = 0
		}
	}

	if = { limit = { has_modifier = dhe_mod_samjeong_unrest_low } remove_modifier = dhe_mod_samjeong_unrest_low }
	if = { limit = { has_modifier = dhe_mod_samjeong_unrest_high } remove_modifier = dhe_mod_samjeong_unrest_high }
	if = { limit = { has_modifier = dhe_mod_samjeong_unrest_severe } remove_modifier = dhe_mod_samjeong_unrest_severe }

	if = {
		limit = { var:dhe_samjeong_unrest_gauge >= 750 }
		add_modifier = { name = dhe_mod_samjeong_unrest_severe months = 2 }
	}
	else_if = {
		limit = { var:dhe_samjeong_unrest_gauge >= 500 }
		add_modifier = { name = dhe_mod_samjeong_unrest_high months = 2 }
	}
	else_if = {
		limit = { var:dhe_samjeong_unrest_gauge >= 250 }
		add_modifier = { name = dhe_mod_samjeong_unrest_low months = 2 }
	}
}

dhe_samjeong_monthly_unrest = {
	dhe_samjeong_init_unrest_gauge = yes

	if = {
		limit = { tax_level = very_low }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -20 }
	}
	else_if = {
		limit = { tax_level = low }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -10 }
	}
	else_if = {
		limit = { tax_level = medium }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 5 }
	}
	else_if = {
		limit = { tax_level = high }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 15 }
	}
	else_if = {
		limit = { tax_level = very_high }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 30 }
	}

	if = {
		limit = { has_law = law_type:law_hwangok_system }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 12 }
	}
	if = {
		limit = { has_law_or_variant = law_type:law_peasant_levies }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 8 }
	}
	if = {
		limit = { bureaucracy < -10 }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 10 }
	}
	if = {
		limit = { government_legitimacy >= 60 }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -8 }
	}
	if = {
		limit = { average_sol < 8 }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 12 }
	}
	if = {
		limit = { average_sol >= 10 }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -6 }
	}
	if = {
		limit = { any_scope_state = { turmoil >= 0.20 } }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 15 }
	}
	if = {
		limit = { ig:ig_landowners = { clout >= 0.25 } }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 8 }
	}

	if = {
		limit = {
			NOT = { has_variable = dhe_samjeong_land_focus }
			NOT = { has_variable = dhe_samjeong_military_focus }
			NOT = { has_variable = dhe_samjeong_granary_focus }
			NOT = { has_variable = dhe_samjeong_compromise_focus }
		}
		change_variable = { name = dhe_samjeong_unrest_gauge add = 8 }
	}
	if = {
		limit = { has_variable = dhe_samjeong_land_focus }
		change_variable = { name = dhe_samjeong_unrest_gauge add = 6 }
	}
	if = {
		limit = { has_variable = dhe_samjeong_military_focus }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -8 }
	}
	if = {
		limit = { has_variable = dhe_samjeong_granary_focus }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -12 }
	}
	if = {
		limit = { has_variable = dhe_samjeong_compromise_focus }
		change_variable = { name = dhe_samjeong_unrest_gauge add = -6 }
	}

	dhe_samjeong_resolve_unrest_gauge = yes
}
