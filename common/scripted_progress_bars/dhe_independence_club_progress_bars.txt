dhe_progress_bar_constitutional_pressure = {
	name = "dhe_progress_bar_constitutional_pressure"
	desc = "dhe_progress_bar_constitutional_pressure_desc"
	default = yes
	double_sided_bad = yes

	start_value = 0
	min_value = -40
	max_value = 40

	monthly_progress = {
		if = {
			limit = { owner = { has_variable = dhe_ic_public_network } }
			add = { desc = "dhe_ic_public_network" value = 1.0 }
		}
		if = {
			limit = { owner = { has_variable = dhe_ic_constitutional_petition } }
			add = { desc = "dhe_ic_constitutional_petition" value = 1.5 }
		}
		if = {
			limit = {
				owner = {
					any_political_movement = {
						is_political_movement_type = movement_dhe_independence_club
						political_movement_support >= 0.05
					}
				}
			}
			add = { desc = "dhe_ic_movement_minor_support" value = 1.0 }
		}
		if = {
			limit = {
				owner = {
					any_political_movement = {
						is_political_movement_type = movement_dhe_independence_club
						political_movement_support >= 0.10
					}
				}
			}
			add = { desc = "dhe_ic_movement_growing_support" value = 1.0 }
		}
		if = {
			limit = {
				owner = {
					any_political_movement = {
						is_political_movement_type = movement_dhe_independence_club
						political_movement_support >= 0.20
					}
				}
			}
			add = { desc = "dhe_ic_movement_mass_support" value = 2.0 }
		}
		if = {
			limit = { owner = { ig:ig_intelligentsia = { clout >= 0.15 } } }
			add = { desc = "dhe_ic_intelligentsia_support" value = 2.0 }
		}
		if = {
			limit = { owner = { literacy_rate >= 0.4 } }
			add = { desc = "dhe_ic_literate_public" value = 1.0 }
		}
		if = {
			limit = { owner = { has_law = law_type:law_protected_speech } }
			add = { desc = "dhe_ic_free_speech" value = 2.0 }
		}
		if = {
			limit = { owner = { government_legitimacy < 50 } }
			add = { desc = "dhe_ic_low_legitimacy" value = 1.0 }
		}
		if = {
			limit = { owner = { has_variable = dhe_ic_palace_guard } }
			subtract = { desc = "dhe_ic_palace_guard" value = 1.0 }
		}
		if = {
			limit = { owner = { has_variable = dhe_ic_royalist_press } }
			subtract = { desc = "dhe_ic_royalist_press" value = 1.5 }
		}
		if = {
			limit = { owner = { has_law = law_type:law_censorship } }
			subtract = { desc = "dhe_ic_censorship" value = 2.0 }
		}
		if = {
			limit = { owner = { government_legitimacy >= 75 } }
			subtract = { desc = "dhe_ic_high_legitimacy" value = 1.0 }
		}

			limit = { owner = { ig:ig_landowners = { is_in_government = yes } } }
