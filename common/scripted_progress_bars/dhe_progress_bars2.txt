	name = "dhe_progress_bar_royal_authority"
	desc = "dhe_progress_bar_royal_authority_desc"
	default = yes

	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {

		# ── [기본] 세도 정치의 관성 ──
		if = {
			limit = { always = yes }
			subtract = { desc = "dhe_royal_auth_base" value = 4.0 }
		}

		# ── [페널티] 지주 clout 30% 이상: 조정 장악 ──
		if = {
			limit = { owner = { ig:ig_landowners = { clout >= 0.3 } } }
			subtract = { desc = "dhe_royal_auth_landowners_dominant" value = 3.0 }
		}

		# ── [페널티] 세습 관료제: 양반 세습 체제 ──
		if = {
			limit = { owner = { has_law = law_type:law_hereditary_bureaucrats } }
			subtract = { desc = "dhe_royal_auth_hereditary" value = 2.0 }
		}

		# ── [페널티] 농노제: 구체제의 무게 ──
		if = {
			limit = { owner = { has_law = law_type:law_serfdom } }
			subtract = { desc = "dhe_royal_auth_serfdom" value = 1.0 }

		if = {
			limit = {
				owner = {
					ig:ig_landowners = { clout < 0.2 }
					NOT = { ig:ig_landowners = { clout < 0.1 } }
				}
			}
			add = { desc = "dhe_royal_auth_landowners_weak" value = 3.0 }
		}

			limit = { owner = { ig:ig_landowners = { clout < 0.1 } } }
			add = { desc = "dhe_royal_auth_landowners_crushed" value = 2.0 }
		}

		# ── [보너스] 정통성 50 이상: 왕권 안정 (★ 고유 핵심) ──
		if = {
			limit = { owner = { government_legitimacy >= 50 } }
			add = { desc = "dhe_royal_auth_legitimacy_stable" value = 2.0 }
		}

		# ── [보너스] 정통성 75 이상: 왕권 절대 ──
		if = {
			limit = { owner = { government_legitimacy >= 75 } }
			add = { desc = "dhe_royal_auth_legitimacy_supreme" value = 1.0 }
		}

			limit = { owner = { ig:ig_intelligentsia = { clout >= 0.15 } } }
