namespace = dhe_japan_control

dhe_japan_control.1 = {
	type = country_event
	placement = root
	title = dhe_japan_control.1.t
	desc = dhe_japan_control.1.d
	flavor = dhe_japan_control.1.f
	icon = "gfx/interface/icons/event_icons/dhe_tokyo_governor.png"

	is_triggered_only = yes

	option = {
		name = dhe_japan_control.1.a
		default_option = yes
		set_variable = dhe_japanese_governor_general_established
		add_prestige = 50
		add_modifier = { name = dhe_mod_japanese_governor_general_active months = 120 }
		c:KOR ?= {
			if = {
				limit = {
					exists = c:JAP
					c:JAP = {
						is_subject = no
					}
				}
				create_diplomatic_pact = {
					country = c:JAP
					type = puppet
				}
			}
			else_if = {
				limit = {
					exists = c:JAP
					c:JAP = {
						is_subject = yes
						NOT = { is_subject_of = c:KOR }
					}
				}
				transfer_subject = c:JAP
			}
			c:JAP ?= {
				if = {
					limit = {
						is_subject = yes
						NOT = { is_subject_type = subject_type_puppet }
					}
					change_subject_type = subject_type_puppet
				}
				set_variable = dhe_japanese_governor_general_country_name
			}
			if = {
				limit = {
					exists = c:EZO
					c:EZO = {
						is_subject = yes
						NOT = { is_subject_of = c:KOR }
					}
				}
				transfer_subject = c:EZO
			}
			else_if = {
				limit = {
					exists = c:EZO
					c:EZO = {
						is_subject = no
					}
				}
				if = {
					limit = { is_country_type = unrecognized }
					create_diplomatic_pact = {
						country = c:EZO
						type = vassal
					}
				}
				else = {
					create_diplomatic_pact = {
						country = c:EZO
						type = puppet
					}
				}
			}
			if = {
				limit = {
					exists = c:RYU
					c:RYU = {
						is_subject = yes
						NOT = { is_subject_of = c:KOR }
					}
				}
				transfer_subject = c:RYU
			}
			else_if = {
				limit = {
					exists = c:RYU
					c:RYU = {
						is_subject = no
					}
				}
				if = {
					limit = { is_country_type = unrecognized }
					create_diplomatic_pact = {
						country = c:RYU
						type = vassal
					}
				}
				else = {
					create_diplomatic_pact = {
						country = c:RYU
						type = puppet
					}
				}
			}
		}
		if = {
			limit = {
				NOT = { has_journal_entry = je_dhe_japanese_governor_general }
			}
			add_journal_entry = { type = je_dhe_japanese_governor_general }
		}
		c:JAP ?= {
			add_modifier = { name = dhe_mod_japanese_governor_general_under_korea months = 120 }
			add_radicals = 50000
			change_relations = {
				country = c:KOR
				value = -20
			}
		}
		if = {
			limit = {
				exists = c:JAP
				c:JAP = {
					is_subject = no
				}
			}
			c:KOR ?= {
				create_diplomatic_pact = {
					country = c:JAP
					type = puppet
				}
			}
		}
		if = {
			limit = {
				exists = c:JAP
				c:JAP = {
					is_subject_of = c:KOR
					NOT = { is_subject_type = subject_type_puppet }
				}
			}
			c:JAP = {
				change_subject_type = subject_type_puppet
			}
		}
		if = {
			limit = {
				exists = c:JAP
				c:JAP = {
					is_subject_of = c:KOR
					is_subject_type = subject_type_puppet
				}
			}
			trigger_event = { id = dhe_japan_control.21 days = 1 popup = yes }
		}
	}
}
