				name = dhe_modifier_jiandao_settlement

			}

			# ==========================================================
            create_pop = {
				culture = korean
				pop_type = peasants   # 직업을 지정 안 하면 튕길 수 있으니 '소작농'으로 고정
				size = 50000          # 5만 명 스폰!
			}
		}
	}
